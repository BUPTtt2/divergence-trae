**把"演"当成一个 Claude Code 风格的 coding agent——只是它的 domain 是"决策推演"而不是"写代码"。**

## 3.1 YanAgent（演）— 自主 LLM 循环

**核心范式**：演 = 一个 Claude Code 风格的 coding agent，只是它"写的代码"是"决策推演的规划"，它"调的工具"是"起卦、召智囊、查记忆"。

```
class YanAgent {
  constructor({ llm, tools, memory, eventBus, trace, autonomyGate }) {
    this.llm = llm;
    this.tools = tools;            // 演的决策语义工具集
    this.memory = memory;          // MemoryStore
    this.eventBus = eventBus;      // 事件总线
    this.trace = trace;            // OpenTelemetry tracer
    this.autonomyGate = autonomyGate;
  }

  async run(session) {
    // 演是 ReAct 循环：Think → Act → Observe → Critique
    while (!session.isDone()) {
      const span = this.trace.startSpan('yan.step', {
        'session.id': session.id,
        'session.iteration': session.iteration,
      });

      try {
        // 1. 感知：读 Blackboard + 记忆 + 历史
        const context = await this.perceive(session);

        // 2. 思考：LLM 决定下一步
        const decision = await this.think(context, session);
        span.setAttribute('decision', decision.action);
        this.eventBus.emit('yan.thinking', decision);

        // 3. 执行：调用工具 / 召智囊
        const result = await this.act(decision, session);
        span.setAttribute('result.type', result.type);

        // 4. 观察：把结果写回 Blackboard
        await this.observe(session, decision, result);
        this.eventBus.emit('yan.observed', { decision, result });

        // 5. 自评：这一步够好吗？要不要重规划？
        const critique = await this.selfCritique(session, decision, result);
        if (critique.replan && session.replanCount < 1) {
          session.replanCount++;
          this.eventBus.emit('replan.triggered', { reason: critique.reason });
          await session.replan(critique.reason);
        }

        // 6. 自主性判定：追问 / 停止 / 继续
        const autonomy = await this.autonomyGate.evaluate(session);
        if (autonomy.action === 'ASK_USER') {
          this.eventBus.emit('user.asked', autonomy.questions);
          await session.pause('WAIT_USER', autonomy.questions);
          break;
        } else if (autonomy.action === 'STOP') {
          await this.finalize(session);
          break;
        }

        // 7. 持久化
        await SessionRepository.save(session);
        session.iteration++;
      } finally {
        span.end();
      }
    }

    return session;
  }

  async think(context, session) {
    const prompt = renderPrompt(YAN_SYSTEM_PROMPT, {
      question: session.question,
      blackboard: context.blackboard.summary(),
      memory: context.memory,
      availableTools: this.tools.list(),
      history: context.history,
      iteration: session.iteration,
      replanCount: session.replanCount,
    });

    const response = await this.trace.traceLLMCall('yan.think', () =&gt;
      this.llm.complete({
        prompt,
        tools: this.tools.specs(),         // 决策语义工具
        responseFormat: YanDecisionSchema,  // Zod schema 强约束
        temperature: 0.3,                  // 决策要稳
        maxTokens: 2000,
      })
    );

    return parseDecision(response);  // 结构化决策
  }
}
```

## 3.3 ZhiguanAgent — 智囊基类

智囊不再是"被叫到就说一句"的 persona——它有自己的工具子集、自己的调用循环、自己的 stance 标注。

```
class ZhiguanAgent {
  constructor({ id, persona, systemPrompt, tools, llm }) {
    this.id = id;
    this.persona = persona;
    this.systemPrompt = systemPrompt;
    this.tools = tools;   // 智囊自己的工具子集（如风眼能调 web_search）
    this.llm = llm;
  }

  async speak({ instruction, context }) {
    const messages = [
      { role: 'system', content: this.renderSystemPrompt() },
      { role: 'user', content: this.renderUserPrompt({ instruction, context }) },
    ];

    // 智囊可调自己的工具（function calling）
    for (let i = 0; i &lt; 3; i++) {  // 智囊最多 3 轮工具调用
      const response = await this.llm.complete({
        messages,
        tools: this.tools.specs(),
        temperature: 0.7,  // 智囊要有人情味
        maxTokens: 300,
      });

      if (!response.toolCalls || response.toolCalls.length === 0) {
        // 智囊发完言
        return {
          agentId: this.id,
          content: response.content,
          stance: this.parseStance(response.content),
          toolUsed: messages.some(m =&gt; m.role === 'tool'),
          confidence: this.estimateConfidence(response.content),
        };
      }

      // 智囊调工具
      for (const call of response.toolCalls) {
        const result = await this.tools.execute(call.name, call.args);
        messages.push({ role: 'tool', name: call.name, content: JSON.stringify(result) });
      }
    }

    // 智囊调太多轮，强制收尾
    return await this.forceConclude(messages);
  }

  renderSystemPrompt() {
    return `你是 ${this.persona.name}（${this.persona.title}）。

## 视角
${this.persona.perspective}

## 方法论
${this.persona.methodology}

## 交付标准
${this.persona.deliverable}

## 边界
- 只能在自己专长领域发言：${this.persona.domain}
- 跨界话题必须明示"这不在我专长内"
- 引用数据时必须用工具验证，不凭印象

## 你能调用的工具
${this.tools.list()}`;
  }
}
```

# 4. 事件总线（Event Bus）— 替代 Blackboard

v1.0 的 Blackboard 是"消息存储 + 序列";v2.0 的 EventBus 是"订阅发布 + 持久化事件流"。区别：

- **订阅发布**：任意 Agent（包括未来加的）可订阅特定事件，动态加入
- **持久化事件流**：所有事件进 PostgreSQL，Session 可重放、可恢复
- **调试可观测**：每个事件都有 timestamp、actor、payload、span context

```
import { EventEmitter } from 'events';

class EventBus {
  constructor({ sessionId, persistence }) {
    this.emitter = new EventEmitter();
    this.sessionId = sessionId;
    this.persistence = persistence;  // SessionRepository
  }

  emit(eventType, payload) {
    const event = {
      id: uuid(),
      sessionId: this.sessionId,
      type: eventType,
      payload,
      timestamp: Date.now(),
    };

    // 1. 发给订阅者
    this.emitter.emit(eventType, event);
    this.emitter.emit('*', event);  // 兜底订阅

    // 2. 持久化（关键：Session 可重放）
    this.persistence.appendEvent(event).catch(err =&gt;
      console.error('EventBus persist failed', err)
    );

    return event;
  }

  // 事件类型
  emitYanThinking(decision) { return this.emit('yan.thinking', decision); }
  emitZhiguanSpoke(finding) { return this.emit('zhiguan.spoke', finding); }
  emitToolCalled(name, result) { return this.emit('tool.called', { name, result }); }
  emitConflictDetected(conflicts) { return this.emit('conflict.detected', conflicts); }
  emitReplanTriggered(reason) { return this.emit('replan.triggered', { reason }); }
  emitUserAsked(questions) { return this.emit('user.asked', { questions }); }
  emitSessionPaused(reason) { return this.emit('session.paused', { reason }); }
  emitSessionComplete(final) { return this.emit('session.complete', final); }

  on(eventType, handler) { this.emitter.on(eventType, handler); }
  off(eventType, handler) { this.emitter.off(eventType, handler); }

  // 重放：从事件流恢复 Session
  async replay() {
    const events = await this.persistence.getEvents(this.sessionId);
    return events;
  }
}
```

# 5. 持久化 Session + Resume

Session 是 v2.0 引入的核心抽象——它是"一次推演"的完整状态对象，存数据库，**可暂停可恢复**。

```
class Session {
  constructor({ id, userId, question }) {
    this.id = id;
    this.userId = userId;
    this.question = question;
    this.status = 'running';  // running | paused | waiting_user | completed | failed
    this.iteration = 0;
    this.replanCount = 0;
    this.events = [];          // 事件流（内存镜像 + DB 持久化）
    this.blackboard = new Blackboard();
    this.plan = null;
    this.toolResults = [];
    this.findings = [];
    this.memoryUsed = [];
    this.pendingQuestions = [];
    this.final = null;
    this.createdAt = Date.now();
    this.updatedAt = Date.now();
  }

  // 关键：从事件流恢复 Session（支持断点续推）
  static async fromEvents(sessionId, repository) {
    const events = await repository.getEvents(sessionId);
    if (events.length === 0) throw new Error('Session not found');

    const session = new Session({
      id: events[0].sessionId,
      userId: events[0].userId,
      question: events[0].question,
    });

    // 重放所有事件
    for (const event of events) {
      session.applyEvent(event);
    }
    return session;
  }

  applyEvent(event) {
    this.events.push(event);
    switch (event.type) {
      case 'yan.thinking':     this.iteration++; break;
      case 'zhiguan.spoke':    this.findings.push(event.payload); break;
      case 'tool.called':      this.toolResults.push(event.payload); break;
      case 'replan.triggered': this.replanCount++; break;
      case 'user.asked':       this.pendingQuestions.push(event.payload); break;
      case 'session.complete': this.status = 'completed'; this.final = event.payload; break;
      case 'session.paused':   this.status = 'paused'; break;
    }
  }

  isDone() { return ['completed', 'failed', 'paused', 'waiting_user'].includes(this.status); }

  async pause(reason) {
    this.status = 'paused';
    await SessionRepository.save(this);
  }

  async resume() {
    this.status = 'running';
    await SessionRepository.save(this);
  }
}
```

**状态**

**含义**

`running`

正在推演

`waiting_user`

演追问，等用户回答

`paused`

用户主动暂停（关 app / 切走）

`completed`

推演完成

`failed`

推演失败（错误恢复后状态）

class MemoryStore {
  constructor({ pg, qdrant, graph }) {
    this.pg = pg;
    this.qdrant = qdrant;
    this.graph = graph;
  }

  // L1: Working Memory - 当前 Session
  async working(sessionId) {
    return this.pg.query(
      'SELECT * FROM events WHERE session_id = $1 ORDER BY timestamp',
      [sessionId]
    );
  }

  // L2: Episodic - 历史推演摘要（向量检索）
  async episodic(userId, query, topK = 5) {
    const embedding = await embed(query);
    return await this.qdrant.search('user_episodes', {
      vector: embedding,
      filter: { userId, type: { in: ['summary', 'decision'] } },
      topK,
      scoreThreshold: 0.7,
    });
  }

  // L3: Semantic - 用户命格（图查询）
  async semantic(userId, key) {
    return await this.graph.query(`
      MATCH (u:User {id: $userId})-[:HAS_TRAIT]-&gt;(t:Trait)
      WHERE t.key CONTAINS $key
      RETURN t
      ORDER BY t.importance DESC
      LIMIT 10
    `, { userId, key });
  }

  // L4: Procedural - 推演技巧
  procedural(intent) {
    return PLAYBOOK[intent] || DEFAULT_PLAYBOOK;
  }

  // Reflect 阶段调用：从本次推演提炼长期记忆
  async consolidate(sessionId) {
    const session = await SessionRepository.load(sessionId);

    // L1 → L2: 摘要
    const summary = await llmSummarize(session);
    await this.pg.query(`
      INSERT INTO session_summaries (session_id, user_id, summary, question, choice, created_at)
      VALUES ($1, $2, $3, $4, $5, NOW())
    `, [sessionId, session.userId, summary, session.question, session.final?.choice]);

    // L2 → L3: 提取命格
    const newTraits = await llmExtractTraits(session, summary);
    for (const trait of newTraits) {
      // 去重：与已有命格向量比对
      const existing = await this.episodic(session.userId, trait.content, 1);
      if (existing[0]?.score &gt; 0.85) {
        // 合并更新
        await this.updateTrait(existing[0].id, trait);
      } else {
        // 新增
        const embedding = await embed(trait.content);
        await this.qdrant.upsert('user_episodes', {
          id: uuid(),
          vector: embedding,
          payload: { userId: session.userId, type: 'trait', ...trait },
        });
      }
    }
  }
}
```

**命格提取的 prompt 关键**：从 Session 提取 3-5 条"用户决策指纹"，例如：

- "用户在风险类决策中倾向保守（连续 3 次选择低风险选项）"
- "用户对'北京租房'有持续关注（3 个月内推演 4 次）"
- "用户决策时优先考虑财务维度，其次是体验"

命格不是事实，是**模式识别**。LLM 提取 + 向量去重 + 人工反馈校准。

```
class AutonomyGate {
  async evaluate(session) {
    // 1. 信息充分性检查
    const sufficiency = await this.checkInfoSufficiency(session);
    if (sufficiency.missing.length &gt; 0 &amp;&amp; session.askCount &lt; 2) {
      return {
        action: 'ASK_USER',
        questions: sufficiency.missing,
        reason: sufficiency.reason,
      };
    }

    // 2. 工具需求检查
    if (this.shouldProbeTools(session) &amp;&amp; !session.toolProbed) {
      return { action: 'PROBE_TOOLS' };
    }

    // 3. 完成度检查
    if (this.isComplete(session)) {
      return { action: 'STOP' };
    }

    // 4. 危险工具检测（HITL）
    if (this.hasDangerousTool(session)) {
      return { action: 'HUMAN_REVIEW' };
    }

    return { action: 'CONTINUE' };
  }

  async checkInfoSufficiency(session) {
    // LLM 评估当前信息是否充分
    const evaluation = await this.llm.evaluate(`
      Question: ${session.question}
      Available info: ${session.blackboard.summary()}
      Memory: ${JSON.stringify(session.memoryUsed)}

      Rate sufficiency 0-1. If &lt; 0.7, what specific info is missing?
    `);

    return {
      missing: evaluation.missingFields,
      reason: evaluation.reason,
    };
  }

  isComplete(session) {
    return (
      session.findings.length &gt;= 3 &amp;&amp;          // 至少 3 个视角
      session.findings.some(f =&gt; f.toolUsed) &amp;&amp;  // 至少 1 个用了工具
      !session.blackboard.hasUnresolvedConflicts()  // 矛盾已处理
    );
  }
}
```

**硬约束（不可违反）**：

- 演追问最多 2 轮（避免无限追问）
- 演可在 Reflect 阶段触发最多 1 次重规划（replanCount ≤ 1）
- 演不能改写用户问题
- 演不能跳过用户抉择直接出最终结论

class HumanInTheLoop {
  // 危险工具拦截
  async interceptDangerousTool(toolCall, session) {
    const DANGEROUS = [
      'send_email',
      'make_payment',
      'post_social_media',
      'delete_data',
    ];

    if (!DANGEROUS.includes(toolCall.name)) return { allowed: true };

    // 弹审批卡
    const approval = await this.sendApprovalCard({
      userId: session.userId,
      title: `演想调用 ${toolCall.name}`,
      description: `参数：${JSON.stringify(toolCall.args)}`,
      reason: toolCall.reason,
    });

    return {
      allowed: approval.approved,
      reason: approval.reason,
    };
  }

  // 质量检查拦截
  async checkQuality(session) {
    const evalResult = await this.eval.runAutoEval(session.id);

    if (evalResult.every(c =&gt; c.passed)) return { passed: true };

    // 质量不达标，让演自检
    const reCritique = await session.engine.selfCritique(session);

    if (reCritique.shouldRedo) {
      return { passed: false, action: 'REDO', reason: reCritique.reason };
    }

    return { passed: false, action: 'USER_TAKE_OVER' };
  }
}

import { trace, context, SpanStatusCode } from '@opentelemetry/api';
import { Langfuse } from 'langfuse';

class Tracer {
  constructor({ serviceName, langfuseKey, langfuseSecret, langfuseHost }) {
    this.tracer = trace.getTracer(serviceName);
    this.langfuse = new Langfuse({
      publicKey: langfuseKey,
      secretKey: langfuseSecret,
      baseUrl: langfuseHost,
    });
  }

  startSpan(name, attributes = {}) {
    const span = this.tracer.startSpan(name, {
      attributes: {
        'service.name': 'yan-deliberation',
        ...attributes,
      },
    });
    return span;
  }

  // LLM 调用专用
  async traceLLMCall(name, fn, metadata = {}) {
    const span = this.startSpan(`llm.${name}`);
    const langfuseTrace = this.langfuse.trace({
      name: `llm.${name}`,
      metadata,
    });

    try {
      const result = await context.with(trace.setSpan(context.active(), span), fn);

      // OpenTelemetry 属性
      span.setAttribute('llm.model', result.model);
      span.setAttribute('llm.tokens.input', result.usage?.input || 0);
      span.setAttribute('llm.tokens.output', result.usage?.output || 0);
      span.setAttribute('llm.cost', result.cost || 0);
      span.setAttribute('llm.latency_ms', result.latencyMs);
      span.setStatus({ code: SpanStatusCode.OK });

      // Langfuse 上报
      langfuseTrace.update({
        output: result.content?.substring(0, 1000),
        usage: {
          input: result.usage?.input || 0,
          output: result.usage?.output || 0,
          total: (result.usage?.input || 0) + (result.usage?.output || 0),
        },
        metadata: { ...metadata, cost: result.cost },
      });

      return result;
    } catch (err) {
      span.recordException(err);
      span.setStatus({ code: SpanStatusCode.ERROR, message: err.message });
      langfuseTrace.update({ statusMessage: err.message });
      throw err;
    } finally {
      span.end();
      await langfuseTrace.flushAsync();
    }
  }
}

```
class EvalPipeline {
  // 自动评估：每次 Session 完成后跑
  async runAutoEval(sessionId) {
    const session = await SessionRepository.load(sessionId);

    const checks = [
      { name: 'no_hallucination',      weight: 0.2, fn: () =&gt; this.checkHallucination(session) },
      { name: 'multi_perspective',     weight: 0.15, fn: async () =&gt; session.findings.length &gt;= 3 },
      { name: 'tool_used_when_needed', weight: 0.1, fn: () =&gt; this.checkToolUsage(session) },
      { name: 'memory_referenced',     weight: 0.1, fn: () =&gt; this.checkMemoryReferenced(session) },
      { name: 'conflict_acknowledged', weight: 0.1, fn: () =&gt; this.checkConflictHandled(session) },
      { name: 'hexagram_consistent',   weight: 0.15, fn: () =&gt; this.checkHexagramConsistency(session) },
      { name: 'stance_diversity',      weight: 0.1, fn: () =&gt; this.checkStanceDiversity(session) },
      { name: 'recommendation_actionable', weight: 0.1, fn: () =&gt; this.checkActionable(session) },
    ];

    const results = await Promise.all(
      checks.map(async c =&gt; ({
        name: c.name,
        weight: c.weight,
        passed: await c.fn(),
        score: 0,
      }))
    );

    const totalScore = results.reduce((s, r) =&gt; s + (r.passed ? r.weight : 0), 0);

    return {
      sessionId,
      totalScore,
      results,
      passed: totalScore &gt;= 0.7,
      evaluatedAt: Date.now(),
    };
  }

  // 人工反馈
  async recordFeedback(sessionId, feedback) {
    await this.pg.query(`
      INSERT INTO user_feedback (session_id, user_id, rating, helpful, issues, created_at)
      VALUES ($1, $2, $3, $4, $5, NOW())
    `, [sessionId, feedback.userId, feedback.rating, feedback.helpful, feedback.issues]);

    // 触发自动评估
    const auto = await this.runAutoEval(sessionId);

    // 校准信号：差评 + 自动评估差 → 标记复盘
    if (feedback.rating &lt;= 2 &amp;&amp; auto.totalScore &lt; 0.6) {
      await this.flagForReview(sessionId, {
        userRating: feedback.rating,
        autoScore: auto.totalScore,
        issues: feedback.issues,
      });
    }

    return auto;
  }
}
